// Memoria dinamica con realloc
definicion_de_tipo estructura {
    sin_signo caracter *d;
    entero n;
    entero cap;
} Big;

estatico vacio *xmalloc(size_t s) { vacio *p = malloc(s);  si (!p) { perror("malloc");  exit(1); } retornar p; }
estatico vacio *xcalloc(size_t n, size_t s) { vacio *p = calloc(n, s); si (!p) { perror("calloc"); exit(1); } retornar p; }
estatico vacio *xrealloc(vacio *p, size_t s) { vacio *q = realloc(p, s); si (!q) { perror("realloc"); exit(1); } retornar q; }

estatico vacio big_ensure(Big *b, entero need) {
    si (need <= b->cap) retornar;
    entero nc = b->cap > 0 ? b->cap : BIG_INIT_CAP;
    mientras (nc < need) nc *= 2;
    b->d = xrealloc(b->d, nc);
    memset(b->d + b->cap, 0, nc - b->cap);
    b->cap = nc;
}

Big *big_new(vacio) {
    Big *b = xmalloc(tamano_de(Big));
    b->cap = BIG_INIT_CAP;
    b->d = xcalloc(b->cap, 1);
    b->n = 1;
    retornar b;
}

Big *big_from_int(entero v) {
    Big *b = big_new();
    si (v == 0) retornar b;
    entero n = 0;
    mientras (v > 0) { big_ensure(b, n + 1); b->d[n++] = v % 10; v /= 10; }
    b->n = n;
    retornar b;
}

Big *big_from_string(constante caracter *s) {
    Big *b = big_new();
    entero len = (entero)strlen(s);
    big_ensure(b, len);
    para (entero i = 0; i < len; i++) b->d[i] = s[len - 1 - i] - '0';
    entero n = len;
    mientras (n > 1 && b->d[n - 1] == 0) n--;
    b->n = n;
    retornar b;
}

vacio big_free(Big *b) { si (!b) retornar; free(b->d); free(b); }

caracter *big_to_string(constante Big *b) {
    caracter *s = xmalloc(b->n + 1);
    para (entero i = 0; i < b->n; i++) s[i] = '0' + b->d[b->n - 1 - i];
    s[b->n] = '\0';
    retornar s;
}

entero big_equals(constante Big *a, constante Big *b) {
    si (a->n != b->n) retornar 0;
    retornar memcmp(a->d, b->d, a->n) == 0;
}

entero big_is_one(constante Big *b) { retornar b->n == 1 && b->d[0] == 1; }

vacio big_add(Big *r, constante Big *a, constante Big *b) {
    entero n = a->n > b->n ? a->n : b->n;
    big_ensure(r, n + 2);
    entero carry = 0;
    para (entero i = 0; i < n; i++) {
        entero s = carry + (i < a->n ? a->d[i] : 0) + (i < b->n ? b->d[i] : 0);
        r->d[i] = s % 10;
        carry = s / 10;
    }
    si (carry) r->d[n++] = carry;
    r->n = n;
}

vacio big_mul_small(Big *r, constante Big *a, entero m) {
    big_ensure(r, a->n + 32);
    largo largo carry = 0;
    para (entero i = 0; i < a->n; i++) {
        largo largo s = carry + (largo largo)a->d[i] * m;
        r->d[i] = s % 10;
        carry = s / 10;
    }
    entero n = a->n;
    mientras (carry) { big_ensure(r, n + 1); r->d[n++] = carry % 10; carry /= 10; }
    r->n = n;
}

vacio big_mul_big(Big *r, constante Big *a, constante Big *b) {
    entero rmax = a->n + b->n + 1;
    sin_signo caracter *res = xcalloc(rmax, 1);
    para (entero i = 0; i < a->n; i++) {
        entero carry = 0, j;
        para (j = 0; j < b->n; j++) {
            entero s = res[i + j] + a->d[i] * b->d[j] + carry;
            res[i + j] = s % 10;
            carry = s / 10;
        }
        entero idx = i + j;
        mientras (carry > 0) {
            entero s = res[idx] + carry;
            res[idx] = s % 10;
            carry = s / 10;
            idx++;
        }
    }
    entero rn = a->n + b->n;
    mientras (rn > 1 && res[rn - 1] == 0) rn--;
    big_ensure(r, rn);
    memcpy(r->d, res, rn);
    r->n = rn;
    free(res);
}

// Triangulo de Pascal (fila n)
Big **pascal_row(entero n) {
    Big **prev = xmalloc(tamano_de(Big *));
    prev[0] = big_from_int(1);
    para (entero r = 1; r <= n; r++) {
        Big **curr = xmalloc(tamano_de(Big *) * (r + 1));
        curr[0] = big_from_int(1);
        curr[r] = big_from_int(1);
        para (entero k = 1; k < r; k++) {
            curr[k] = big_new();
            big_add(curr[k], prev[k - 1], prev[k]);
        }
        para (entero k = 0; k < r; k++) big_free(prev[k]);
        free(prev);
        prev = curr;
    }
    retornar prev;
}

vacio free_row(Big **row, entero n) {
    para (entero k = 0; k <= n; k++) big_free(row[k]);
    free(row);
}

Big *evaluate(Big **coef, entero n, entero x) {
    Big *acc = big_from_int(0);
    Big *pow_x = big_from_int(1);
    para (entero k = 0; k <= n; k++) {
        Big *term = big_new();
        big_mul_big(term, coef[k], pow_x);
        Big *new_acc = big_new();
        big_add(new_acc, acc, term);
        big_free(acc); big_free(term);
        acc = new_acc;
        si (k < n) {
            Big *np = big_new();
            big_mul_small(np, pow_x, x);
            big_free(pow_x);
            pow_x = np;
        }
    }
    big_free(pow_x);
    retornar acc;
}

caracter *polynomial_string(Big **coef, entero n) {
    entero total = n + 1;
    caracter **terms = (caracter **)xmalloc(tamano_de(caracter *) * total);
    size_t len = 0;
    para (entero k = n; k >= 0; k--) {
        caracter *cs = big_to_string(coef[k]);
        caracter var[24];
        si (k == 0) var[0] = '\0';
        si_no si (k == 1) strcpy(var, "x");
        si_no sprintf(var, "x^%d", k);
        entero omit_one = (k > 0 && big_is_one(coef[k]));
        caracter *t = (caracter *)xmalloc(strlen(cs) + strlen(var) + 2);
        si (omit_one) sprintf(t, "%s", var);
        si_no sprintf(t, "%s%s", cs, var);
        terms[n - k] = t;
        len += strlen(t) + 3;
        free(cs);
    }
    caracter *out = (caracter *)xmalloc(len + 1);
    out[0] = '\0';
    para (entero i = 0; i < total; i++) {
        si (i) strcat(out, " + ");
        strcat(out, terms[i]);

    }
    free(terms);
    retornar out;
}
